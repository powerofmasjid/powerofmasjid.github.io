const CACHE_NAME = 'pom-admin-cache-v2'; const urlsToCache = [ '/', '/admin.html', '/superadmin.html', '/manifest.json', '/Logo Power of Masjid.png' ]; self.addEventListener('install', event => { event.waitUntil( caches.open(CACHE_NAME) .then(cache => { // Pakai add() satu-satu (bukan addAll) supaya kalau 1 file gagal (404/offline), // file lain tetap ter-cache, tidak menggagalkan instalasi service worker seluruhnya. return Promise.all( urlsToCache.map(url => cache.add(url).catch(err => { console.log('SW: gagal cache', url, err); })) ); }) ); self.skipWaiting(); }); self.addEventListener('activate', event => { // Bersihkan cache versi lama agar tidak menumpuk & tidak menyajikan file basi event.waitUntil( caches.keys().then(names => Promise.all( names.filter(n => n !== CACHE_NAME).map(n => caches.delete(n)) )) ); self.clients.claim(); }); self.addEventListener('fetch', event => { event.respondWith( caches.match(event.request) .then(response => { if (response) { return response; } return fetch(event.request); }) ); });
